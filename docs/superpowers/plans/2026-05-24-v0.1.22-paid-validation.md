```python
"""Merge probe + main + safety reports into v0.1.22-prod.md canonical."""

from pathlib import Path
import re

PROBE = Path("evals/reports/v0.1.22/probe.md")
MAIN = Path("evals/reports/v0.1.22/v0.1.22-prod-main.md")
SAFETY = Path("evals/reports/v0.1.22/v0.1.22-prod-safety-adhoc.md")
OUT = Path("evals/reports/v0.1.22/v0.1.22-prod.md")


def extract_case_blocks(report: Path) -> list[str]:
    """Extract per-case Markdown blocks starting `### Case \`...\``"""
    text = report.read_text(encoding="utf-8")
    blocks = re.split(r"(?=^### Case `)", text, flags=re.MULTILINE)
    return [b for b in blocks if b.startswith("### Case `")]


def extract_header(report: Path) -> str:
    """Extract everything before the first case block (header + aggregate)."""
    text = report.read_text(encoding="utf-8")
    match = re.search(r"^### Case `", text, flags=re.MULTILINE)
    return text[: match.start()] if match else text


def main() -> None:
    probe_cases = extract_case_blocks(PROBE)
    main_cases = extract_case_blocks(MAIN)
    safety_cases = extract_case_blocks(SAFETY)
    header = extract_header(MAIN)  # Use main header (most representative)
    merged = header + "\n".join(probe_cases + main_cases + safety_cases)
    OUT.write_text(merged, encoding="utf-8")
    print(f"Merged {len(probe_cases) + len(main_cases) + len(safety_cases)} cases into {OUT}")


if __name__ == "__main__":
    main()
```
