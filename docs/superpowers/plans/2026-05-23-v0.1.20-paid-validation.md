### Step 9: Create `scripts/v0120_run.py`

Use Write tool. Verbatim content:

```python
"""v0.1.20 — paid validation runner (mirrors scripts/h15_run.py).

Routes harness report output to evals/reports/v0.1.20/<tag>.md after each run,
then restores the committed canonical evals/reports/latest.md (Path-B isolation
pattern from h15_run.py + ab_eval.py). The harness writes its report
unconditionally to evals/reports/latest.md; we snapshot+restore so the
canonical baseline is never left clobbered by a paid v0.1.20 run.

USER-GATED: real runs cost Anthropic credit; invoke only on explicit OK after
the spec's SKIP/PROCEED gate (T3) authorizes. Invoke as:

  uv run --env-file .env python -m scripts.v0120_run --version v1.0 \\
      --cases-file evals/v0120_probe_a_ids.txt --tag armA-probe

Both arms run via SEPARATE invocations (full process restart between) per
spec §4 apples-to-apples controls.
"""

from __future__ import annotations

import argparse
import os
import subprocess
from pathlib import Path

from evals.harness import _REPORT_PATH
from evals.harness import main as _harness_main

_V0120_DIR = Path("evals/reports/v0.1.20")


def _load_case_ids(cases_file: Path) -> set[str]:
    """BOM-safe, comment/blank-skipping id-set loader (h15_run.py precedent)."""
    return {
        ln.strip()
        for ln in cases_file.read_text(encoding="utf-8-sig").splitlines()
        if ln.strip() and not ln.strip().startswith("#")
    }


def _isolate_report(tag: str) -> None:
    """Path-B isolation: snapshot this run's report to evals/reports/v0.1.20/<tag>.md,
    then restore the committed canonical evals/reports/latest.md.

    Mirrors scripts/h15_run.py::_isolate_report exactly. If the harness process
    is hard-killed between writing latest.md and this call, recover the
    committed baseline with `git checkout HEAD -- evals/reports/latest.md`.
    """
    _V0120_DIR.mkdir(parents=True, exist_ok=True)
    dest = _V0120_DIR / f"{tag}.md"
    if _REPORT_PATH.exists():
        dest.write_bytes(_REPORT_PATH.read_bytes())
    subprocess.run(["git", "checkout", "HEAD", "--", str(_REPORT_PATH)], check=True, timeout=30)


def run(*, version: str, cases_file: Path, tag: str) -> None:
    """Run the harness for ONE arm with env routing, snapshot, restore.

    version: "v1.0" (ARM A; baseline) or "v1.4" (ARM B; candidate).
    cases_file: allowlist (probe 5 ids OR main 59 ids).
    tag: report basename under evals/reports/v0.1.20/ (e.g. "armA-probe").
    """
    ids = _load_case_ids(cases_file)
    prev = os.environ.get("REGULAITOR_ANALYST_PROMPT_VERSION")
    os.environ["REGULAITOR_ANALYST_PROMPT_VERSION"] = version
    try:
        try:
            # subset=None: the case_ids allowlist is the truth source. NO --subset
            # truncation (would silently drop cases beyond the count).
            _harness_main(subset=None, case_ids=ids)
        finally:
            # Snapshot + restore canonical latest.md even if the (paid) harness
            # crashed mid/post-run. h15_run.py precedent: this exception (if
            # any) replaces the original — acceptable per ab_eval.py pattern.
            _isolate_report(tag)
    finally:
        if prev is None:
            os.environ.pop("REGULAITOR_ANALYST_PROMPT_VERSION", None)
        else:
            os.environ["REGULAITOR_ANALYST_PROMPT_VERSION"] = prev


def parse_args() -> argparse.Namespace:
    p = argparse.ArgumentParser(description="v0.1.20 USER-GATED paid validation runner")
    p.add_argument(
        "--version",
        required=True,
        choices=["v1.0", "v1.4"],
        help="Analyst prompt version (v1.0 ARM A baseline / v1.4 ARM B candidate)",
    )
    p.add_argument(
        "--cases-file",
        type=Path,
        required=True,
        help="Newline-delimited case-id allowlist (e.g. evals/v0120_probe_a_ids.txt)",
    )
    p.add_argument(
        "--tag",
        required=True,
        help="Report basename under evals/reports/v0.1.20/ (e.g. armA-probe, armB-main)",
    )
    return p.parse_args()


if __name__ == "__main__":
    a = parse_args()
    run(version=a.version, cases_file=a.cases_file, tag=a.tag)
```

"""$0 unit tests for scripts/v0120_run.py — verify env routing + isolation contract.

Mirrors tests/unit/scripts/test_h15_run.py if it exists; otherwise these tests
provide the same coverage shape: id loader, env set/restore, isolate stubbed.

from __future__ import annotations

import os
from pathlib import Path

import pytest

from scripts import v0120_run


def test_load_case_ids_strips_comments_and_blanks(tmp_path: Path) -> None:
    f = tmp_path / "ids.txt"
    f.write_text("# header\nchat-001\n\nchat-002\n  # indented comment\nchat-003\n", encoding="utf-8")
    ids = v0120_run._load_case_ids(f)
    assert ids == {"chat-001", "chat-002", "chat-003"}


def test_load_case_ids_handles_utf8_bom(tmp_path: Path) -> None:
    f = tmp_path / "ids.txt"
    # UTF-8 BOM + ids
    f.write_bytes(b"\xef\xbb\xbf# header\nchat-001\nchat-002\n")
    ids = v0120_run._load_case_ids(f)
    assert ids == {"chat-001", "chat-002"}


def test_run_sets_env_then_restores_prior(tmp_path: Path, monkeypatch: pytest.MonkeyPatch) -> None:
    # Prior env value: ensure it's restored, not lost.
    monkeypatch.setenv("REGULAITOR_ANALYST_PROMPT_VERSION", "v1.3")
    # Mock the harness + isolation to no-ops; we test ONLY the env contract.
    observed: dict[str, str | None] = {}

    def fake_harness(*, subset: int | None, case_ids: set[str] | None) -> None:
        observed["during_run"] = os.environ.get("REGULAITOR_ANALYST_PROMPT_VERSION")

    monkeypatch.setattr(v0120_run, "_harness_main", fake_harness)
    monkeypatch.setattr(v0120_run, "_isolate_report", lambda tag: None)

    cases_file = tmp_path / "ids.txt"
    cases_file.write_text("chat-001\n", encoding="utf-8")

    v0120_run.run(version="v1.4", cases_file=cases_file, tag="test-tag")

    assert observed["during_run"] == "v1.4"  # env was set during harness call
    # Env was restored to prior value, not lost or set to v1.4.
    assert os.environ.get("REGULAITOR_ANALYST_PROMPT_VERSION") == "v1.3"


def test_run_sets_env_then_unsets_when_prior_unset(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    # Prior env value: UNSET. After run, env must remain UNSET (not stuck at v1.4).
    monkeypatch.delenv("REGULAITOR_ANALYST_PROMPT_VERSION", raising=False)

    monkeypatch.setattr(v0120_run, "_harness_main", lambda *, subset, case_ids: None)
    monkeypatch.setattr(v0120_run, "_isolate_report", lambda tag: None)

    cases_file = tmp_path / "ids.txt"
    cases_file.write_text("chat-001\n", encoding="utf-8")

    v0120_run.run(version="v1.0", cases_file=cases_file, tag="test-tag")

    assert "REGULAITOR_ANALYST_PROMPT_VERSION" not in os.environ


def test_run_restores_env_even_on_harness_exception(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    monkeypatch.setenv("REGULAITOR_ANALYST_PROMPT_VERSION", "v1.2")

    def crashy_harness(*, subset: int | None, case_ids: set[str] | None) -> None:
        raise RuntimeError("simulated mid-run crash")

    monkeypatch.setattr(v0120_run, "_harness_main", crashy_harness)
    # _isolate_report still called per finally; mock it to no-op so it doesn't
    # try to git-checkout in the test.
    monkeypatch.setattr(v0120_run, "_isolate_report", lambda tag: None)

    cases_file = tmp_path / "ids.txt"
    cases_file.write_text("chat-001\n", encoding="utf-8")

    with pytest.raises(RuntimeError, match="simulated mid-run crash"):
        v0120_run.run(version="v1.4", cases_file=cases_file, tag="test-tag")

    # Env STILL restored even though harness raised.
    assert os.environ.get("REGULAITOR_ANALYST_PROMPT_VERSION") == "v1.2"

def _load_arm_from_checkpoint(checkpoint_root: Path, run_ids: list[str]) -> list[ChatCaseResult | DocCaseResult]:
    """Concatenate per-case results across N run_ids (probe + main).

    Each call to evals.harness.main() generates a distinct run_id; an arm's
    full 64-case result requires loading + concatenating the 5-case probe
    checkpoint + the 59-case main checkpoint.
    """
    all_cases: list[ChatCaseResult | DocCaseResult] = []
    for rid in run_ids:
        all_cases.extend(checkpoint.load_completed(rid, root=checkpoint_root))
    return all_cases


def _subset_chat(cases: list[ChatCaseResult | DocCaseResult], ids: set[str]) -> list[ChatCaseResult]:
    """Subset to ChatCaseResult instances whose case_id is in ids."""
    return [c for c in cases if isinstance(c, ChatCaseResult) and c.case_id in ids]


def _format_metric_delta(a: float, b: float, bar: float | None) -> str:
    delta = b - a
    sign = "+" if delta >= 0 else ""
    bar_check = ""
    if bar is not None:
        a_pass = "PASS" if a >= bar else "FAIL"
        b_pass = "PASS" if b >= bar else "FAIL"
        bar_check = f" (bar {bar:.2f}: A {a_pass} / B {b_pass})"
    return f"A={a:.3f}, B={b:.3f}, delta={sign}{delta:.3f}{bar_check}"


def emit_unified_arm_report(
    arm_label: str,
    probe_path: Path,
    main_path: Path,
    output_path: Path,
) -> None:
    """Concatenate probe + main per-case rows into a unified arm report.

    String-surgery on the markdown: keeps the harness's exact table format
    so the comparison script (and any future reader) can parse it the same
    way. Aggregate footer is regenerated from the union.
    """
    probe_text = probe_path.read_text(encoding="utf-8")
    main_text = main_path.read_text(encoding="utf-8")

    # Header from probe (run timestamp etc); body = concatenated per-case rows.
    # The harness writes "## Chat results (per case)" as a section header;
    # we splice main's section body after probe's.
    header_marker = "## Chat results (per case)"
    if header_marker not in probe_text or header_marker not in main_text:
        raise ValueError(
            f"Cannot find {header_marker!r} in arm reports; harness format may have changed"
        )

    probe_header, _, probe_body = probe_text.partition(header_marker)
    main_header, _, main_body = main_text.partition(header_marker)

    # The aggregate footer (everything before "## Chat results (per case)") is
    # regenerated by re-rendering from the concatenated case list. For v0.1.20
    # we keep the probe's header verbatim + concatenate the table rows so the
    # per-case appendix is complete. The aggregate metrics MAY drift slightly
    # from a true 64-case re-aggregation — the authoritative aggregate is in
    # comparison.md (re-aggregated from checkpoint).
    output_path.write_text(
        probe_header + header_marker + probe_body + "\n\n## --- Main run rows ---\n" + main_body,
        encoding="utf-8",
    )


def emit_comparison_report(
    arm_a_cases: list[ChatCaseResult],
    arm_b_cases: list[ChatCaseResult],
    output_path: Path,
    cost_total_a: float,
    cost_total_b: float,
    forecast_total: float,
) -> None:
    """Per-metric A/B comparison on the v0.1.20-bar 30-case subset + per-cohort breakdown."""
    # H10 30-case subset comparison (the bar measurement)
    a_h10 = _subset_chat(arm_a_cases, _H10_30_CASE_IDS)
    b_h10 = _subset_chat(arm_b_cases, _H10_30_CASE_IDS)

    if len(a_h10) != len(b_h10):
        raise ValueError(
            f"H10 subset count mismatch: A={len(a_h10)} B={len(b_h10)}; "
            "cohorts must be aligned for apples-to-apples comparison"
        )

    a_agg_h10 = aggregate(a_h10, [])
    b_agg_h10 = aggregate(b_h10, [])

    lines: list[str] = [
        "# v0.1.20 ARM A (v1.0) vs ARM B (v1.4) — comparison report",
        "",
        f"Cohort: 64 chat cases (probe 5 + main 59); H10 subset = first 30.",
        "",
        "## H10 30-case subset — ADR-0021 v0.1.20-bar comparison",
        "",
        "| Metric | ARM A (v1.0) | ARM B (v1.4) | Delta | v0.1.20-bar | A pass? | B pass? |",
        "|---|---|---|---|---|---|---|",
    ]

    for metric_attr, bar in [
        ("faithfulness_mean", _V0120_BAR["faithfulness"]),
        ("answer_relevancy_mean", _V0120_BAR["answer_relevancy"]),
        ("context_precision_mean", _V0120_BAR["context_precision"]),
        ("citation_precision_mean", _V0120_BAR["citation_precision"]),
        ("citation_recall_mean", _V0120_BAR["citation_recall"]),
        ("verdict_match_rate", _V0120_BAR["verdict_match"]),
        ("severity_match_rate", _V0120_BAR["severity_match"]),
    ]:
        a_val = getattr(a_agg_h10, metric_attr)
        b_val = getattr(b_agg_h10, metric_attr)
        delta = b_val - a_val
        sign = "+" if delta >= 0 else ""
        a_pass = "PASS" if a_val >= bar else "FAIL"
        b_pass = "PASS" if b_val >= bar else "FAIL"
        metric_name = metric_attr.replace("_mean", "").replace("_rate", "")
        lines.append(
            f"| {metric_name} | {a_val:.3f} | {b_val:.3f} | {sign}{delta:.3f} | "
            f"{bar:.2f} | {a_pass} | {b_pass} |"
        )

    lines.extend(["", "## Per-cohort exploratory breakdown (no bar)"])

    for cohort_label, cohort_ids in [
        ("H14 cross-corpus (14 cases)", _H14_CROSS_CORPUS_IDS),
        ("v0.1.13 industry (10 cases)", _V0113_INDUSTRY_IDS),
        ("v0.1.15 gap-analysis (10 cases)", _V0115_GAP_IDS),
    ]:
        a_sub = _subset_chat(arm_a_cases, cohort_ids)
        b_sub = _subset_chat(arm_b_cases, cohort_ids)
        if not a_sub or not b_sub:
            lines.append(f"\n### {cohort_label}\n\n(empty subset; skipping)")
            continue
        a_agg = aggregate(a_sub, [])
        b_agg = aggregate(b_sub, [])
        lines.extend([
            f"\n### {cohort_label}",
            "",
            "| Metric | ARM A (v1.0) | ARM B (v1.4) | Delta |",
            "|---|---|---|---|",
        ])
        for metric_attr in [
            "faithfulness_mean", "answer_relevancy_mean", "context_precision_mean",
            "citation_precision_mean", "citation_recall_mean",
            "verdict_match_rate", "severity_match_rate",
        ]:
            a_val = getattr(a_agg, metric_attr)
            b_val = getattr(b_agg, metric_attr)
            delta = b_val - a_val
            sign = "+" if delta >= 0 else ""
            metric_name = metric_attr.replace("_mean", "").replace("_rate", "")
            lines.append(
                f"| {metric_name} | {a_val:.3f} | {b_val:.3f} | {sign}{delta:.3f} |"
            )

    lines.extend([
        "",
        "## Cost",
        "",
        f"- ARM A total (probe + main): €{cost_total_a:.2f}",
        f"- ARM B total (probe + main): €{cost_total_b:.2f}",
        f"- Combined actual: €{cost_total_a + cost_total_b:.2f}",
        f"- T3 forecast (×1.5 high): €{forecast_total:.2f}",
        f"- Variance: {((cost_total_a + cost_total_b) / forecast_total - 1) * 100:+.1f}%" if forecast_total > 0 else "",
    ])

    output_path.write_text("\n".join(lines) + "\n", encoding="utf-8")


def parse_args() -> argparse.Namespace:
    p = argparse.ArgumentParser(description="v0.1.20 A/B comparison report generator")
    p.add_argument("--arm-a-probe-run-id", required=True, help="run_id for ARM A probe checkpoint")
    p.add_argument("--arm-a-main-run-id", required=True, help="run_id for ARM A main checkpoint")

    p.add_argument("--forecast", type=float, required=True, help="T3 forecast total EUR (×1.5 high)")

def main() -> None:
    a = parse_args()

### D1 — Variant matrix: 1-dim {v1.0 vs v1.4} = 2 arms

ARM A baseline (`REGULAITOR_ANALYST_PROMPT_VERSION` unset → v1.0 default per `src/regulaitor/agents/analyst.py:69`). ARM B candidate (env=v1.4 → v0.1.17.1 prompt loaded). Both arms IDENTICAL otherwise: Sonnet 4.6 production model, Haiku 4.5 judge, production default retrieval config, Council binding ON, MonotonicEscalatePolicy, same gold set + time window.

### D2 — Cohort: 64 chat (skip doc)

All 64 chat gold cases × 2 arms = 128 paid Analyst calls. Doc-mode (10 cases) SKIPPED honestly per §22.22 design-coherence catch: `prompts/document_analyst/` contains ONLY `system.v1.0.md`. Setting env=v1.4 in ARM B would crash every doc case at `prompt_path.read_text()`. Running doc cases with v1.0 in BOTH arms gives 0 A/B signal at 2× cost. Doc-mode A/B carried forward as separate future milestone.

Primary reporting: H10 baseline 30 cases against ADR-0021 v0.1.20-bar. Exploratory: H14 cross-corpus 14 + v0.1.13 industry 10 + v0.1.15 gap-analysis 10. Exploratory NOT used as flip-decision input (out of ADR-0021 bar scope).

### D3 — Cost gate: strict 2-probe

Spec §3 D3 four-step protocol: PROBE-A (5 v1.0 cases) → PROBE-B (5 v1.4 cases) → extrapolate `total_high = (mean_a + mean_b) × 64 × 1.5` → SKIP/PROCEED gate. Probe cases are the first 5 entries of each arm's cohort; NOT additional. Main run is chat-006..064 (59 cases per arm); T6 comparison script concatenates probe + main into the unified 64-case reports.

**Spec amendment documented honestly (§22.22):** the spec assumed checkpoint resume across two `evals.harness.main()` invocations. Plan-writing T0 discovered each invocation generates a distinct `run_id`; two invocations produce TWO checkpoint files. No new `--start_at` flag was added (avoids backend touch during measurement milestone). Resume implemented via disjoint case-id allowlists. T6 `scripts/v0120_compare.py` reads both checkpoint files + concatenates per-case rows + the two arm markdown reports.

### D4 — Flip protocol: hard safety floor + soft narrative

## Alternatives considered (rejected)

1. **2-dim factorial {v1.0, v1.4} × {default retrieval, per-norma cap=2}** = 4 arms. **Rejected**: 2× cost; per-norma cap measurement deferred to separate future milestone.
2. **3-dim factorial adding Council ON/OFF** = 8 arms. **Rejected**: 4× cost; ADR-0025 D2 rejected env override for Council binding as YAGNI; scope creep.

4. **Single-probe variant** (just probe A, infer B). **Rejected**: spec §3 D3 critical-path argues both arms must probe for cost-drift detection (v1.4 verbosity might explode cost).
5. **Include doc cases with v1.0 in both arms**. **Rejected**: 0 A/B signal at 2× doc cost; design-incoherent.
6. **Backport v1.4 to `prompts/document_analyst/` for doc-mode A/B coverage**. **Rejected**: scope creep (would require its own prompt-version skill review + tests + decisions); doc-mode A/B carried forward as separate future milestone.
